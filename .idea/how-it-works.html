Сниппет кода

// Псевдо-обработчик заявки (Go)
type Application struct {
    Name  string
    Age   int
    Phone string
    Track string // "trekking" | "wrestling" | "media"
}

func Validate(a Application) error {
    if a.Age < 13 || a.Age > 17 { return fmt.Errorf("доступно 13–17 лет") }
    if a.Name == "" { return fmt.Errorf("нужно имя") }
    return nil
}
